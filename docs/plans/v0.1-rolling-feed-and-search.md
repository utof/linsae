# linsae v0.1 — rolling feed and search — implementation plan

**Status:** rev 2 (Opus plan-review pass 1 applied, 2026-05-26).

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. **Read `docs/specs/v0.1-rolling-feed-and-search.md` and `CLAUDE.md` once, then this plan top to bottom before starting Task 1.** Cite spec line numbers in commit messages where the change implements a specific section.

**Goal:** A working, tested, lefthook-gated Electron note-taking app whose only surfaces are a Telegram-style chronological feed of markdown notes (with KaTeX math), `?`-promote-to-question, `[[wikilink]]` resolution + backlinks pane, `⌘K` full-text search, and a `Reveal notes folder` menu item. Notes are markdown files on disk; SQLite is an index.

**Architecture:** Main process owns SQLite (`better-sqlite3` + Drizzle migrations + raw prepared statements) and the file system; renderer is sandboxed and talks to main via typed IPC channels validated by Zod. Markdown files in `app.getPath('userData')/notes/<uuid>.md` are the source of truth. SQLite is rebuilt from disk by the startup reconciler. Write atomicity = tmp + fsync + rename, then SQLite upsert in one transaction.

**Tech stack:** Electron 30+, electron-vite, React 19, TypeScript strict, better-sqlite3 (raw SQL migrations via Vite `?raw` glob imports — no Drizzle at v0.1), vanilla `Virtuoso` (MIT — not the commercial `VirtuosoMessageList`), `cmdk`, `react-markdown` + `remark-gfm` + `remark-math` + `rehype-katex`, KaTeX, `lucide-react`, `react-hotkeys-hook`, `@tanstack/react-query`, `uuidv7`, `js-yaml`, Vitest + Testing Library + jsdom, Biome, knip, lefthook, pnpm. No Tailwind at v0.1 — components use inline `style` props with v21 CSS-variable tokens directly.

**Branch:** `phase/v0.1-rolling-feed-and-search`. (CLAUDE.md uses the prefix `phase/` literally for branch naming; the slug after it is a milestone name.)

```tsx
      get:    vi.fn(async () => null),
```

```ts
export function normalizeSlug(raw: string): string {
  return raw.trim().toLowerCase().replace(/\s+/g, ' ')
}

export function slugFromBody(body: string): string {
  const firstNonEmptyLine = body
    .split('\n')
    .map(l => l.trim())
    .find(l => l.length > 0)
  if (!firstNonEmptyLine) return ''
  const stripped = firstNonEmptyLine.replace(/^#+\s*/, '')
  return normalizeSlug(stripped)
}
```

```ts
import { normalizeSlug } from './slug'

export interface Wikilink {
  slug: string         // normalized lowercase trimmed target
  display: string      // visible text (= target if no |alias)
  section: string | null  // verbatim section after # (no normalization)
  raw: string          // the entire [[...]] match for replacement
}

const WIKILINK_RE = /\[\[([^\]]+)\]\]/g

export function extractWikilinks(body: string): Wikilink[] {
  const out: Wikilink[] = []
  for (const match of body.matchAll(WIKILINK_RE)) {
    const inner = match[1]
    const [beforePipe, afterPipe] = inner.split('|', 2)
    const display = (afterPipe ?? beforePipe).trim()
    const [targetPart, sectionPart] = beforePipe.split('#', 2)
    const slug = normalizeSlug(targetPart)
    if (!slug) continue
    out.push({
      slug,
      display: display.length > 0 ? display : targetPart.trim(),
      section: sectionPart?.trim() || null,
      raw: match[0],
    })
  }
  return out
}
```

```ts
import yaml from 'js-yaml'
import type { NoteType } from '../../shared/types'

export interface NoteFrontmatter {
  id: string
  slug: string
  type: NoteType
  created_at: number
  updated_at: number
  deleted_at?: number
  aliases?: string[]
}

export type ParseResult =
  | { ok: true; frontmatter: NoteFrontmatter; body: string }
  | { ok: false; error: string }

const DELIM = '---'

export function parseFrontmatter(file: string): ParseResult {
  if (!file.startsWith(DELIM)) return { ok: false, error: 'no frontmatter delimiter at start' }
  const rest = file.slice(DELIM.length).split('\n').slice(1).join('\n')
  const end = rest.indexOf(`\n${DELIM}`)
  if (end === -1) return { ok: false, error: 'no closing frontmatter delimiter' }
  const yamlText = rest.slice(0, end)
  const body = rest.slice(end + DELIM.length + 1).replace(/^\n/, '')
  try {
    const fm = yaml.load(yamlText) as NoteFrontmatter
    if (!fm || typeof fm.id !== 'string' || typeof fm.slug !== 'string') {
      return { ok: false, error: 'frontmatter missing required fields (id, slug)' }
    }
    return { ok: true, frontmatter: fm, body }
  } catch (e) {
    return { ok: false, error: `yaml parse error: ${(e as Error).message}` }
  }
}

export function serializeFrontmatter(fm: NoteFrontmatter, body: string): string {
  const yamlText = yaml.dump(fm, { lineWidth: 100, noRefs: true }).trimEnd()
  return `---\n${yamlText}\n---\n\n${body}`
}
```

```ts
import { openSync, fsyncSync, writeSync, closeSync, renameSync, unlinkSync } from 'node:fs'

export function atomicWriteFile(targetPath: string, contents: string): void {
  const tmpPath = `${targetPath}.tmp`
  const buf = Buffer.from(contents, 'utf8')
  let fd: number | null = null
  try {
    fd = openSync(tmpPath, 'w', 0o644)
    writeSync(fd, buf, 0, buf.length, 0)
    fsyncSync(fd)
    closeSync(fd)
    fd = null
    renameSync(tmpPath, targetPath)
  } catch (e) {
    if (fd !== null) {
      try { closeSync(fd) } catch { /* ignore */ }
    }
    try { unlinkSync(tmpPath) } catch { /* ignore */ }
    throw e
  }
}
```

```ts
import { readdirSync, mkdirSync, existsSync, readFileSync, statSync } from 'node:fs'
import { join } from 'node:path'
import { atomicWriteFile } from './atomic-write'
import { parseFrontmatter, serializeFrontmatter, type NoteFrontmatter, type ParseResult } from './frontmatter'

export class NotesDir {
  constructor(private readonly dir: string) {
    if (!existsSync(dir)) mkdirSync(dir, { recursive: true })
  }

  private pathFor(id: string): string {
    return join(this.dir, `${id}.md`)
  }

  writeNote(fm: NoteFrontmatter, body: string): void {
    atomicWriteFile(this.pathFor(fm.id), serializeFrontmatter(fm, body))
  }

  readNote(id: string): ParseResult {
    const p = this.pathFor(id)
    if (!existsSync(p)) return { ok: false, error: 'file does not exist' }
    const raw = readFileSync(p, 'utf8')
    return parseFrontmatter(raw)
  }

  listNoteIds(): string[] {
    return readdirSync(this.dir)
      .filter(name => name.endsWith('.md'))
      .map(name => name.slice(0, -3))
  }

  statNote(id: string): { mtimeMs: number; size: number } | null {
    const p = this.pathFor(id)
    if (!existsSync(p)) return null
    const s = statSync(p)
    return { mtimeMs: s.mtimeMs, size: s.size }
  }
}
```

```ts
export function openDb(path: string): DB {
```

- [ ] **Step 5: Implement `src/main/db/migrate.ts` using Vite `?raw` glob imports**

Vite's `publicDir` does NOT copy assets adjacent to the compiled main bundle (verified against electron-vite docs in plan review A12). Reading `migrations/*.sql` at runtime via `fileURLToPath(import.meta.url)` resolves to `out/main/` in production, where `migrations/` doesn't exist → ENOENT crash. The robust pattern is `import.meta.glob('./migrations/*.sql', { query: '?raw', import: 'default', eager: true })` so Vite inlines the SQL contents as strings at build time.

```ts
import type { Database as DB } from 'better-sqlite3'

// Vite resolves each ./migrations/*.sql to its raw string contents at build time.
// Add new migrations by dropping <NNNN>_<name>.sql next to this file — no other change needed.
const migrations = import.meta.glob('./migrations/*.sql', {
  query: '?raw',
  import: 'default',
  eager: true,
}) as Record<string, string>

export function runMigrations(db: DB): void {
  db.exec(`CREATE TABLE IF NOT EXISTS _migrations (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL UNIQUE,
    applied_at INTEGER NOT NULL
  )`)

  const applied = new Set(
    (db.prepare('SELECT name FROM _migrations').all() as { name: string }[]).map(r => r.name),
  )

  const entries = Object.entries(migrations)
    .map(([path, sql]) => [path.split('/').pop()!, sql] as const)
    .sort(([a], [b]) => a.localeCompare(b))

  for (const [name, sql] of entries) {
    if (applied.has(name)) continue
    const apply = db.transaction(() => {
      db.exec(sql)
      db.prepare('INSERT INTO _migrations (name, applied_at) VALUES (?, ?)').run(name, Date.now())
    })
    apply()
  }
}
```

Note: the migration sql for `0001_init.sql` already includes `CREATE TABLE _migrations` — the `IF NOT EXISTS` in `runMigrations` plus the idempotent migration-name check make this safe across re-runs.

```ts
  return db.prepare(`SELECT id, slug, body, type, created_at, updated_at, deleted_at

}

}
```

```ts
import type { Database as DB } from 'better-sqlite3'
import type { Note, SearchHit } from '../../../shared/types'

// FTS5 treats `"`, `(`, `)`, `:`, `*`, `^` etc. as query operators. Raw user
// input like `O'Hara` or `f(x)` produces a syntax error. Wrap the user's query
// as a literal phrase by double-quoting and escaping any embedded double-quotes.
// This sacrifices OR/AND query syntax — acceptable for v0.1 where users expect
// substring-like search.
function ftsPhraseEscape(raw: string): string {
  return '"' + raw.replace(/"/g, '""') + '"'
}

export function searchNotes(db: DB, opts: { query: string; limit: number }): SearchHit[] {
  // FTS5 uses SQLite's implicit INTEGER rowid; join via notes.rowid = notes_fts.rowid
  let rows: Array<Note & { snippet: string; rank: number }>
  try {
    rows = db.prepare(
      `SELECT n.id, n.slug, n.body, n.type, n.created_at, n.updated_at, n.deleted_at,
              snippet(notes_fts, 0, '<mark>', '</mark>', '…', 16) AS snippet,
              bm25(notes_fts) AS rank
       FROM notes_fts
       JOIN notes n ON n.rowid = notes_fts.rowid
       WHERE notes_fts MATCH ?
         AND n.deleted_at IS NULL
       ORDER BY rank
       LIMIT ?`,
    ).all(ftsPhraseEscape(opts.query), opts.limit) as Array<Note & { snippet: string; rank: number }>
  } catch {
    // Defense in depth: if FTS5 still rejects the escaped phrase, return empty
    // rather than propagate the throw to the renderer.
    return []
  }

  return rows.map(r => ({
    note: {
      id: r.id, slug: r.slug, body: r.body, type: r.type,
      created_at: r.created_at, updated_at: r.updated_at, deleted_at: r.deleted_at,
    },
    snippet: r.snippet,
    rank: r.rank,
  }))
}
```

```ts
import type { Database as DB } from 'better-sqlite3'
import type { NoteType } from '../../../shared/types'

export interface Revision {
  id: string
  note_id: string
  body: string
  type: NoteType
  saved_at: number
  supersedes: string | null
}

export function appendRevision(
  db: DB,
  input: { revisionId: string; noteId: string; body: string; type: NoteType },
): Revision {
  const prev = (db.prepare(
    'SELECT id FROM note_revisions WHERE note_id = ? ORDER BY saved_at DESC LIMIT 1',
  ).get(input.noteId) as { id: string } | undefined)?.id ?? null

  const now = Date.now()
  db.prepare(
    `INSERT INTO note_revisions (id, note_id, body, type, saved_at, supersedes)
     VALUES (?, ?, ?, ?, ?, ?)`,
  ).run(input.revisionId, input.noteId, input.body, input.type, now, prev)

  return {
    id: input.revisionId, note_id: input.noteId, body: input.body, type: input.type,
    saved_at: now, supersedes: prev,
  }
}

export function listRevisions(db: DB, noteId: string): Revision[] {
  return db.prepare(
    `SELECT id, note_id, body, type, saved_at, supersedes
     FROM note_revisions WHERE note_id = ? ORDER BY saved_at DESC`,
  ).all(noteId) as Revision[]
}
```

- [ ] **Step 3: Implement** — file write FIRST, then DB transaction (spec §Storage architecture / Write atomicity; plan-review B1)

The spec is explicit: write the file (tmp + fsync + rename) before any DB mutation. If the process dies between the two, the file on disk is the truth and the next startup reconciliation re-derives the SQLite row. Doing this backwards (DB first) means a crash leaves a DB row that the reconciler then marks `deleted_at`, silently destroying the just-saved note.

Also: on update, preserve any unknown frontmatter keys (including `aliases:`) by reading the existing file first and merging.

```ts
import { uuidv7 } from 'uuidv7'
import type { Database as DB } from 'better-sqlite3'
import type { Note, NoteType } from '../shared/types'
import { NotesDir } from './files/notes-dir'
import { createNote, updateNote, getNote } from './db/queries/notes'
import { replaceLinksForNote } from './db/queries/links'
import { appendRevision } from './db/queries/revisions'
import { extractWikilinks } from './text/wikilinks'
import { slugFromBody } from './text/slug'
import type { NoteFrontmatter } from './files/frontmatter'

type SaveInput =
  | { mode: 'create'; body: string; type: NoteType }
  | { mode: 'update'; id: string; body: string; type: NoteType }
  | { mode: 'softDelete'; id: string }

export function saveNote(db: DB, nd: NotesDir, input: SaveInput): Note {
  const now = Date.now()

  if (input.mode === 'softDelete') {
    const note = getNote(db, input.id)
    if (!note) throw new Error(`note not found: ${input.id}`)

    // Merge with any existing frontmatter (preserves aliases:, etc.)
    const existing = nd.readNote(input.id)
    const baseFm: NoteFrontmatter = existing.ok
      ? { ...existing.frontmatter, updated_at: note.updated_at, deleted_at: now }
      : {
          id: note.id, slug: note.slug, type: note.type,
          created_at: note.created_at, updated_at: note.updated_at, deleted_at: now,
        }

    // 1. File first
    nd.writeNote(baseFm, note.body)

    // 2. DB second, in one transaction
    db.transaction(() => {
      replaceLinksForNote(db, input.id, [])
      db.prepare('UPDATE notes SET deleted_at = ? WHERE id = ?').run(now, input.id)
    })()

    return getNote(db, input.id)!
  }

  // create or update
  const id = input.mode === 'create' ? uuidv7() : input.id
  const existing = input.mode === 'update' ? getNote(db, input.id) : null
  if (input.mode === 'update' && !existing) throw new Error(`note not found: ${input.id}`)

  const slug = input.mode === 'create' ? (slugFromBody(input.body) || id) : existing!.slug
  const created_at = input.mode === 'create' ? now : existing!.created_at
  const links = extractWikilinks(input.body)

  // Merge frontmatter on update to preserve aliases: and any unknown keys
  const existingFile = input.mode === 'update' ? nd.readNote(input.id) : null
  const fm: NoteFrontmatter =
    existingFile && existingFile.ok
      ? { ...existingFile.frontmatter, slug, type: input.type, created_at, updated_at: now,
          deleted_at: undefined }
      : { id, slug, type: input.type, created_at, updated_at: now }

  // 1. File first (atomic tmp + fsync + rename inside nd.writeNote)
  nd.writeNote(fm, input.body)

  // 2. DB second, in one transaction
  const note = db.transaction(() => {
    let n: Note
    if (input.mode === 'create') {
      // createNote and updateNote both call Date.now() internally; threading `now` keeps
      // the file frontmatter and DB row timestamps aligned within the same save.
      db.prepare(`INSERT INTO notes (id, slug, body, type, created_at, updated_at)
                  VALUES (?, ?, ?, ?, ?, ?)`)
        .run(id, slug, input.body, input.type, created_at, now)
      n = getNote(db, id)!
    } else {
      db.prepare(`UPDATE notes SET body = ?, type = ?, updated_at = ? WHERE id = ?`)
        .run(input.body, input.type, now, input.id)
      n = getNote(db, input.id)!
    }
    replaceLinksForNote(db, n.id, links)
    appendRevision(db, { revisionId: uuidv7(), noteId: n.id, body: n.body, type: n.type })
    return n
  })()

  return note
}
```

Note: this bypasses `createNote`/`updateNote` from `src/main/db/queries/notes.ts` for the actual write so we can thread an explicit `now`. The queries-layer functions remain useful for reads (`getNote`, `listNotes`) and for tests that don't go through `saveNote`. `softDeleteNote` is no longer called by `saveNote` — the inline `UPDATE` above replaces it — but keep the export so existing tests still link.

```ts
import { createHash } from 'node:crypto'
import { appendFileSync, existsSync, mkdirSync } from 'node:fs'
import { join } from 'node:path'
import type { Database as DB } from 'better-sqlite3'
import type { ReconcileReport } from '../../shared/types'
import { NotesDir } from '../files/notes-dir'
import { extractWikilinks } from '../text/wikilinks'

function hash(body: string): string {
  return createHash('sha256').update(body).digest('hex')
}

export function reconcile(db: DB, nd: NotesDir, logsDir?: string): ReconcileReport {
  const onDisk = new Set(nd.listNoteIds())
  const report: ReconcileReport = { scanned: onDisk.size, inserted: 0, updated: 0, deleted: 0, skipped: 0 }
  const skippedFiles: Array<{ id: string; error: string }> = []

  db.transaction(() => {
    const existing = new Map<string, { body: string }>()
    for (const row of db.prepare('SELECT id, body FROM notes').all() as { id: string; body: string }[]) {
      existing.set(row.id, { body: row.body })
    }

    for (const id of onDisk) {
      const r = nd.readNote(id)
      if (!r.ok) {
        report.skipped++
        skippedFiles.push({ id, error: r.error })
        continue
      }
      const fm = r.frontmatter
      const prev = existing.get(id)
      if (!prev) {
        db.prepare(`INSERT INTO notes (id, slug, body, type, created_at, updated_at, deleted_at)
                    VALUES (?, ?, ?, ?, ?, ?, ?)`)
          .run(id, fm.slug, r.body, fm.type, fm.created_at, fm.updated_at, fm.deleted_at ?? null)
        const links = extractWikilinks(r.body)
        const insertLink = db.prepare(
          `INSERT OR IGNORE INTO links (from_note_id, to_slug, edge_type) VALUES (?, ?, 'reference')`,
        )
        for (const l of links) insertLink.run(id, l.slug)
        report.inserted++
      } else if (hash(prev.body) !== hash(r.body)) {
        db.prepare(`UPDATE notes SET body = ?, slug = ?, type = ?, updated_at = ?, deleted_at = ?
                    WHERE id = ?`)
          .run(r.body, fm.slug, fm.type, fm.updated_at, fm.deleted_at ?? null, id)
        db.prepare('DELETE FROM links WHERE from_note_id = ?').run(id)
        const links = extractWikilinks(r.body)
        const insertLink = db.prepare(
          `INSERT OR IGNORE INTO links (from_note_id, to_slug, edge_type) VALUES (?, ?, 'reference')`,
        )
        for (const l of links) insertLink.run(id, l.slug)
        report.updated++
      }
    }

    for (const [id] of existing) {
      if (!onDisk.has(id)) {
        db.prepare('UPDATE notes SET deleted_at = ? WHERE id = ?').run(Date.now(), id)
        report.deleted++
      }
    }
  })()

  if (logsDir && skippedFiles.length > 0) {
    if (!existsSync(logsDir)) mkdirSync(logsDir, { recursive: true })
    const stamp = new Date().toISOString()
    const lines = skippedFiles.map(s => `${stamp}\t${s.id}\t${s.error}`).join('\n') + '\n'
    appendFileSync(join(logsDir, 'reconcile.log'), lines)
  }

  return report
}
```

```ts
import { visit } from 'unist-util-visit'
import type { Plugin } from 'unified'
import type { Root, Text } from 'mdast'

const WIKILINK_RE = /\[\[([^\]]+)\]\]/g

interface WikilinkNode {
  type: 'wikilink'
  data: { hName: 'a'; hProperties: { className: string[]; 'data-slug': string; href: string } }
  children: [{ type: 'text'; value: string }]
}

export const remarkWikilinks: Plugin<[], Root> = () => (tree) => {
  visit(tree, 'text', (node: Text, index, parent) => {
    if (!parent || index == null) return
    const matches = [...node.value.matchAll(WIKILINK_RE)]
    if (matches.length === 0) return

    const out: Array<Text | WikilinkNode> = []
    let lastEnd = 0
    for (const m of matches) {
      const start = m.index!
      if (start > lastEnd) out.push({ type: 'text', value: node.value.slice(lastEnd, start) })
      const inner = m[1]
      const [beforePipe, afterPipe] = inner.split('|', 2)
      const display = (afterPipe ?? beforePipe).trim()
      const [targetPart] = beforePipe.split('#', 2)
      const slug = targetPart.trim().toLowerCase().replace(/\s+/g, ' ')
      out.push({
        type: 'wikilink',
        data: { hName: 'a', hProperties: {
          className: ['wikilink'],
          'data-slug': slug,
          href: `linsae://note/${encodeURIComponent(slug)}`,
        } },
        children: [{ type: 'text', value: display.length > 0 ? display : targetPart.trim() }],
      })
      lastEnd = start + m[0].length
    }
    if (lastEnd < node.value.length) out.push({ type: 'text', value: node.value.slice(lastEnd) })
    parent.children.splice(index, 1, ...(out as unknown as Text[]))
    return index + out.length
  })
}
```

```tsx
import { memo, useCallback } from 'react'
import ReactMarkdown from 'react-markdown'
import remarkGfm from 'remark-gfm'
import remarkMath from 'remark-math'
import rehypeKatex from 'rehype-katex'
import { remarkWikilinks } from './remark-wikilinks'

interface Props {
  body: string
  onWikilinkClick: (slug: string) => void
  resolveSlug?: (slug: string) => boolean
}

export const Markdown = memo(function Markdown({ body, onWikilinkClick, resolveSlug }: Props) {
  const handleClick = useCallback((e: React.MouseEvent) => {
    const target = e.target as HTMLElement
    if (target.classList.contains('wikilink')) {
      e.preventDefault()
      const slug = target.dataset.slug
      if (slug) onWikilinkClick(slug)
    }
  }, [onWikilinkClick])

  // After the markdown renders, mark dangling wikilinks. Done via a ref-callback
  // rather than rebuilding the plugin tree so resolveSlug can change without busting
  // the markdown cache.
  const containerRef = useCallback((el: HTMLDivElement | null) => {
    if (!el || !resolveSlug) return
    for (const a of el.querySelectorAll<HTMLAnchorElement>('a.wikilink')) {
      const slug = a.dataset.slug
      if (!slug) continue
      const dangling = !resolveSlug(slug)
      a.classList.toggle('dangling', dangling)
      a.title = dangling ? 'not a note yet — click to start one.' : ''
      a.style.color = dangling ? 'var(--type-question)' : 'var(--accent)'
    }
  }, [resolveSlug, body])

  return (
    <div ref={containerRef} onClick={handleClick}>
      <ReactMarkdown
        remarkPlugins={[remarkGfm, remarkMath, remarkWikilinks]}
        rehypePlugins={[rehypeKatex]}
      >
        {body}
      </ReactMarkdown>
    </div>
  )
})
```

```tsx
import { useState, useRef, useEffect } from 'react'
import type { NoteType } from '../../../shared/types'

interface Props {
  onSubmit: (input: { body: string; type: NoteType }) => void
  onCancel: () => void
  initialBody: string
  initialMode: NoteType
  editMode?: boolean
}

export function Composer({ onSubmit, onCancel, initialBody, initialMode, editMode }: Props) {
  const [body, setBody] = useState(initialBody)
  const [mode, setMode] = useState<NoteType>(initialMode)
  const ref = useRef<HTMLTextAreaElement>(null)

  useEffect(() => { ref.current?.focus() }, [])

  const submit = () => {
    if (!body.trim()) return
    onSubmit({ body, type: mode })
    setBody('')
    setMode('claim')
  }

  const onKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
    if (e.key === '?' && body.length === 0 && mode === 'claim' && !e.metaKey && !e.ctrlKey) {
      e.preventDefault()
      setMode('question')
      return
    }
    if (e.key === 'Escape') {
      // stopPropagation is critical: the global window keydown handler in App.tsx
      // also listens for Esc (to close palette/backlinks). Without it, a single Esc
      // could execute two precedence steps (clear question mode + close backlinks),
      // violating spec §Esc precedence.
      if (mode === 'question') {
        e.preventDefault()
        e.stopPropagation()
        setMode('claim')
        return
      }
      if (editMode) {
        e.preventDefault()
        e.stopPropagation()
        onCancel()
        return
      }
      // else: let the global handler handle it (palette / backlinks / no-op)
    }
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault()
      submit()
    }
  }
```

## Task 30 — Wire the full `App` shell

**Files:**
- Modify: `src/renderer/src/App.tsx`

This is where everything composes. State: notes list (from query), focusedId, paletteOpen, editingNoteId.

- [ ] **Step 1: Implement** — uses `react-hotkeys-hook` for global shortcuts, server-side IPC for alias-aware wikilink resolution, draft-body pattern for dangling links, and a reconcile-skip banner

```tsx
import { useState, useMemo } from 'react'
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { useHotkeys } from 'react-hotkeys-hook'
import { api } from './lib/api'
import { Topbar } from './topbar/Topbar'
import { Feed } from './feed/Feed'
import { Composer } from './composer/Composer'
import { CommandPalette } from './palette/CommandPalette'
import { BacklinksPane } from './backlinks/BacklinksPane'
import type { Note, NoteType } from '../../shared/types'

export function App() {
  const queryClient = useQueryClient()
  const { data: notes = [] } = useQuery({
    queryKey: ['notes'],
    queryFn: () => api.notes.list({ limit: 5000 }),
  })
  const [focusedId, setFocusedId] = useState<string | null>(null)
  const [paletteOpen, setPaletteOpen] = useState(false)
  const [editingNoteId, setEditingNoteId] = useState<string | null>(null)
  const [draftBody, setDraftBody] = useState<string | null>(null)
  const [skipBannerDismissed, setSkipBannerDismissed] = useState(false)

  const slugSet = useMemo(
    () => new Set(notes.filter(n => !n.deleted_at).map(n => n.slug)),
    [notes],
  )
  // Synchronous slug-only resolver for the Markdown component's dangling-class pass.
  // The full alias-aware resolver runs only on click (below) via IPC.
  const resolveSlug = (slug: string) => slugSet.has(slug.toLowerCase().trim())

  const invalidate = () => queryClient.invalidateQueries({ queryKey: ['notes'] })

  const createMut = useMutation({
    mutationFn: ({ body, type }: { body: string; type: NoteType }) => api.notes.create(body, type),
    onSuccess: () => { invalidate(); setDraftBody(null) },
  })
  const updateMut = useMutation({
    mutationFn: ({ id, body, type }: { id: string; body: string; type: NoteType }) =>
      api.notes.update(id, body, type),
    onSuccess: () => { invalidate(); setEditingNoteId(null) },
  })
  const deleteMut = useMutation({
    mutationFn: (id: string) => api.notes.delete(id),
    onSuccess: invalidate,
  })

  // Global shortcuts via react-hotkeys-hook. `enableOnFormTags` lets ⌘K fire from
  // inside the textarea. The composer's local Esc handler calls stopPropagation
  // when it handles Esc itself, so these only fire when composer doesn't.
  useHotkeys('mod+k', e => { e.preventDefault(); setPaletteOpen(o => !o) },
    { enableOnFormTags: ['textarea', 'input'] })
  useHotkeys('esc', () => {
    if (paletteOpen) { setPaletteOpen(false); return }
    if (focusedId)   { setFocusedId(null);  return }
  }, { enableOnFormTags: ['textarea', 'input'] }, [paletteOpen, focusedId])

  const onWikilinkClick = async (slug: string) => {
    // Server-side resolver handles slug + alias + most-recent-wins (spec §Wikilinks).
    const match = await api.links.resolve(slug)
    if (match) {
      setFocusedId(match.id)
      return
    }
    // Dangling: prefill the composer with `# <slug>\n\n` (spec §Wikilinks step 4)
    // — do NOT auto-create. The user must hit Enter to commit, so they can edit
    // the title or back out before any file/DB write happens.
    setDraftBody(`# ${slug}\n\n`)
  }

  const editingNote: Note | undefined = editingNoteId
    ? notes.find(n => n.id === editingNoteId)
    : undefined

  // Reconcile-skip banner: read from the env var set by main process at startup.
  // (Wired by Task 21 as `process.env.LINSAE_RECONCILE_SKIPPED` — main reads its
  // own process.env; renderer doesn't see it directly, so expose via a constant
  // injected by electron-vite's define plugin, OR — simpler — fetch via an IPC.
  // For v0.1, fetch via a one-shot useQuery against system:openLogsFolder-adjacent
  // channel. Keep it simple: pass via a query.)
  const { data: skipped = 0 } = useQuery({
    queryKey: ['reconcile-skipped'],
    queryFn: () => api.system.getReconcileSkipped(),
    staleTime: Infinity,
  })

  return (
    <div style={{ display: 'flex', height: '100vh', overflow: 'hidden', background: 'var(--bg-0)' }}>
      <main style={{ display: 'flex', flexDirection: 'column', flex: 1, minWidth: 0, minHeight: 0 }}>
        <Topbar onOpenPalette={() => setPaletteOpen(true)} />
        {skipped > 0 && !skipBannerDismissed && (
          <div style={{
            padding: '8px 16px', background: '#FDECEC', borderBottom: '1px solid #FAEAC2',
            fontSize: 13, color: 'var(--fg-1)', display: 'flex', gap: 12, alignItems: 'center',
          }}>
            <span>{skipped} notes had unreadable frontmatter and were skipped.</span>
            <button onClick={() => api.system.openLogsFolder()}
              style={{ border: 0, background: 'transparent', color: 'var(--accent)', cursor: 'pointer', textDecoration: 'underline' }}>
              view log.
            </button>
            <div style={{ flex: 1 }} />
            <button onClick={() => setSkipBannerDismissed(true)}
              style={{ border: 0, background: 'transparent', cursor: 'pointer', color: 'var(--fg-2)' }}>dismiss</button>
          </div>
        )}
        {notes.length === 0 ? (
          <div style={{ flex: 1, display: 'flex', alignItems: 'center', justifyContent: 'center',
                        color: 'var(--fg-3)', fontFamily: 'var(--font-sans)', fontSize: 14 }}>
            nothing yet. start anywhere.
          </div>
        ) : (
          <Feed
            notes={notes}
            focusedId={focusedId}
            onFocus={setFocusedId}
            onWikilinkClick={onWikilinkClick}
            resolveSlug={resolveSlug}
            onEdit={setEditingNoteId}
            onDelete={(id) => { deleteMut.mutate(id); if (focusedId === id) setFocusedId(null) }}
            onCopyLink={(id) => navigator.clipboard.writeText(`linsae://note/${id}`)}
          />
        )}
        {editingNote ? (
          <Composer
            key={editingNote.id}
            initialBody={editingNote.body}
            initialMode={editingNote.type}
            editMode
            onSubmit={({ body, type }) => updateMut.mutate({ id: editingNote.id, body, type })}
            onCancel={() => setEditingNoteId(null)}
          />
        ) : (
          <Composer
            key={draftBody ?? 'fresh'}
            initialBody={draftBody ?? ''}
            initialMode="claim"
            onSubmit={({ body, type }) => createMut.mutate({ body, type })}
            onCancel={() => setDraftBody(null)}
          />
        )}
      </main>
      {focusedId && (
        <BacklinksPane
          focusedNoteId={focusedId}
          onClose={() => setFocusedId(null)}
          onJump={setFocusedId}
        />
      )}
      <CommandPalette
        open={paletteOpen}
        onClose={() => setPaletteOpen(false)}
        onJump={setFocusedId}
      />
    </div>
  )
}
```

```ts
import { mkdtempSync, rmSync, readFileSync, existsSync } from 'node:fs'

import { openDb } from '../../src/main/db/client'

import { NotesDir } from '../../src/main/files/notes-dir'
import { reconcile } from '../../src/main/db/reconcile'

let dir: string
let dbPath: string
beforeEach(() => {
  dir = mkdtempSync(join(tmpdir(), 'linsae-int-'))
  dbPath = join(dir, 'db.sqlite')
})

describe('file↔DB round trip', () => {
  it('create note → file on disk + DB row + revision + reconcile is no-op', () => {
    const notesDir = join(dir, 'notes')
    const nd = new NotesDir(notesDir)
    const db = openDb(dbPath)
    runMigrations(db)

    const n = saveNote(db, nd, { mode: 'create', body: '# Foo\n\nbody', type: 'claim' })
    expect(existsSync(join(notesDir, `${n.id}.md`))).toBe(true)

    const report = reconcile(db, nd)
    expect(report.inserted + report.updated + report.deleted).toBe(0)

    db.close()
  })

  it('restart: re-open DB, run reconciler → state preserved', () => {
    const notesDir = join(dir, 'notes')
    let nd = new NotesDir(notesDir)
    let db = openDb(dbPath)
    runMigrations(db)
    saveNote(db, nd, { mode: 'create', body: 'hello', type: 'claim' })
    db.close()

    nd = new NotesDir(notesDir)
    db = openDb(dbPath)
    runMigrations(db)
    reconcile(db, nd)
    expect(db.prepare('SELECT COUNT(*) AS c FROM notes').get()).toEqual({ c: 1 })
    db.close()
  })

  it('orphan recovery: delete DB, restart → reconciler rebuilds from disk', () => {
    const notesDir = join(dir, 'notes')
    let nd = new NotesDir(notesDir)
```
