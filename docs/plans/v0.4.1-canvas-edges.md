**Goal:** Turn the canvas's read-only edge *display* (v0.4 §11) into edge *creation + interaction*: drag card→card to draw a typed `links` row, label it, select it, delete it — and drag into empty space to create-or-connect a note in one gesture.

**Architecture:** Drawn edges are real `links` rows with a NEW `edge_type` (`'link'` plain / free-text label), slug-based like wikilinks. `'reference'`+`'comment-on'` are RESERVED (Zod refusal) so a drawn edge never collides with the `replaceLinksForNote` `'reference'`-wipe. Read side (`canvasEdges`) is unchanged except for one additive `toSlug` field (the PK component delete needs). Rendering extends the existing single `UnderlayLayer`; creation is a new gesture in `useCanvasInteractions`; the target picker + the existing `/` picker share a hand-rolled subsequence `fuzzy.ts`.

2. **Modifier remap** (behavior change, ADR-noted): additive selection becomes **`shift || meta`** (drop `ctrl`) at `useCanvasInteractions.ts:162` and `:193`. **`ctrl`-drag from a card body = plain edge; `ctrl+alt`-drag = labeled edge.** The hover **connect handle** always starts an edge regardless of modifier (the discoverable path). 3. **`ctrl+alt` label field — `esc` cancels the whole edge** (no edge created). A plain edge is one `ctrl`-drag away; this is the least-surprising fallback. `↵` commits the typed label as `edgeType`.
4. **No shared picker base yet.** Add `EdgeTargetPicker.tsx` as its own component (two pickers don't justify a `PickerBase`; revisit on a third — survey + spec agree).
5. **Edge ops are NOT in the §13 layout-undo stack** (spec §7). The drop-in-empty *place* still records a `'place'` op (existing `createMut`/`placeNote` behavior); the edge *create* does not. Documented undo wrinkle (spec §4).

```ts
/**
 * Drawn-edge create/delete. Drawn edges are `links` rows with a non-reserved
 * `edge_type` ('link' or a free-text label) — so `replaceLinksForNote`
 * (deletes only edge_type='reference', links.ts) never wipes them. Slug-based
 * like wikilinks (spec §1). Reserved types are guarded here AND at the Zod
 * boundary (zod-schemas.ts) — defence in depth.
 * @see docs/specs/v0.4.1-canvas-edges.md §1 §2
 */
import type { DB } from '../db'

/** edge_types managed by other subsystems; never valid drawn-edge labels. */
export const RESERVED_EDGE_TYPES = ['reference', 'comment-on'] as const

function liveSlug(db: DB, noteId: string): string | null {
  const r = db.prepare(`SELECT slug FROM notes WHERE id=? AND deleted_at IS NULL`).get(noteId) as
    | { slug: string }
    | undefined
  return r?.slug ?? null
}

export function createDrawnEdge(
  db: DB,
  i: { fromNoteId: string; toNoteId: string; edgeType: string },
): void {
  const type = i.edgeType.trim()
  if (type.length === 0) throw new Error('edgeType must be non-empty')
  if ((RESERVED_EDGE_TYPES as readonly string[]).includes(type))
    throw new Error(`edgeType '${type}' is reserved`)
  db.transaction(() => {
    const fromSlug = liveSlug(db, i.fromNoteId)
    const toSlug = liveSlug(db, i.toNoteId)
    if (!fromSlug || !toSlug) throw new Error('both endpoints must be live notes')
    if (i.fromNoteId === i.toNoteId) throw new Error('no self-edges')
    db.prepare(
      `INSERT OR IGNORE INTO links (from_note_id, to_slug, edge_type) VALUES (?, ?, ?)`,
    ).run(i.fromNoteId, toSlug, type)
  })()
}

export function deleteDrawnEdge(
  db: DB,
  i: { fromNoteId: string; toSlug: string; edgeType: string },
): void {
  if ((RESERVED_EDGE_TYPES as readonly string[]).includes(i.edgeType))
    throw new Error(`edgeType '${i.edgeType}' is read-only on the canvas`)
  db.prepare(`DELETE FROM links WHERE from_note_id=? AND to_slug=? AND edge_type=?`).run(
    i.fromNoteId,
    i.toSlug,
    i.edgeType,
  )
}
```

```ts
/**
 * Hand-rolled case-insensitive subsequence fuzzy matcher (no dep, spec §9).
 * Scores consecutive runs + word-boundary hits higher (fzf-style), returns
 * descending score with matched-char indices for highlight. Used by the canvas
 * `/` picker and the edge-target picker; feed/global search keeps FTS (#130).
 * @see docs/specs/v0.4.1-canvas-edges.md §4
 */
export interface FuzzyResult {
  id: string
  score: number
  /** indices into the ORIGINAL title where query chars matched (for <mark>). */
  matched: number[]
}

function scoreOne(query: string, title: string): { score: number; matched: number[] } | null {
  const q = query.toLowerCase()
  const t = title.toLowerCase()
  const matched: number[] = []
  let qi = 0
  let score = 0
  let prevIdx = -2
  for (let ti = 0; ti < t.length && qi < q.length; ti++) {
    if (t[ti] !== q[qi]) continue
    matched.push(ti)
    // consecutive-run bonus
    if (ti === prevIdx + 1) score += 5
    // word-boundary bonus (start, or after a space/punctuation)
    if (ti === 0 || /[\s\-_/]/.test(t[ti - 1])) score += 10
    score += 1
    prevIdx = ti
    qi++
  }
  if (qi < q.length) return null // not a full subsequence
  // shorter titles rank slightly higher for the same match
  score -= title.length * 0.01
  return { score, matched }
}

export function fuzzyMatch(
  query: string,
  candidates: ReadonlyArray<{ id: string; title: string }>,
): FuzzyResult[] {
  if (query.trim().length === 0)
    return candidates.map((c) => ({ id: c.id, score: 0, matched: [] }))
  const out: FuzzyResult[] = []
  for (const c of candidates) {
    const m = scoreOne(query, c.title)
    if (m) out.push({ id: c.id, score: m.score, matched: m.matched })
  }
  out.sort((a, b) => b.score - a.score)
  return out
}
```

```ts
/** Distance from a world point to the nearest point on a segment (spec §5 hit-test). */
export function pointToSegmentDistance(p: { x: number; y: number }, s: Segment): number {
  const dx = s.x2 - s.x1
  const dy = s.y2 - s.y1
  const lenSq = dx * dx + dy * dy
  if (lenSq === 0) return Math.hypot(p.x - s.x1, p.y - s.y1)
  let t = ((p.x - s.x1) * dx + (p.y - s.y1) * dy) / lenSq
  t = Math.max(0, Math.min(1, t))
  return Math.hypot(p.x - (s.x1 + t * dx), p.y - (s.y1 + t * dy))
}

export interface Arrowhead {
  tip: { x: number; y: number }
  left: { x: number; y: number }
  right: { x: number; y: number }
}
/** Two barb points behind the target tip, for a directed drawn edge (spec §6). `size` in world px. */
export function arrowhead(s: Segment, size: number): Arrowhead {
  const dx = s.x2 - s.x1
  const dy = s.y2 - s.y1
  const len = Math.hypot(dx, dy) || 1
  const ux = dx / len
  const uy = dy / len // unit along the segment toward the tip
  const back = size
  const spread = size * 0.6
  const bx = s.x2 - ux * back
  const by = s.y2 - uy * back
  // perpendicular = (-uy, ux)
  return {
    tip: { x: s.x2, y: s.y2 },
    left: { x: bx - uy * spread, y: by + ux * spread },
    right: { x: bx + uy * spread, y: by - ux * spread },
  }
}
```
